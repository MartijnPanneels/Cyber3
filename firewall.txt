# this is to reset all the previous firewall-rules
flush ruleset

table inet filter {
    chain input {
        type filter hook input priority 0; policy drop;  # Block all incoming

        ct state established,related accept  # Allow ongoing connections
        iif lo accept  # Allow loopback traffic
        ip protocol icmp accept
        tcp dport {22,2222} accept
        
        # Allow ESP (IPsec) packets for tunnel decryption
        ip saddr 192.168.62.42 ip daddr 192.168.62.253 ip protocol esp accept
        ip saddr 192.168.62.253 ip daddr 192.168.62.42 ip protocol esp accept
    }

 chain forward {
        type filter hook forward priority 0; policy drop;

        # Always allow established/related connections first
        ct state established,related accept

        # DNS forwarding
        ip saddr 172.30.128.7 ip daddr 192.168.62.254 udp dport 53 accept
        ip saddr 172.30.128.7 ip daddr 192.168.62.254 tcp dport 53 accept
        ip saddr 192.168.62.254 ip daddr 172.30.128.7 udp sport 53 accept
        ip saddr 192.168.62.254 ip daddr 172.30.128.7 tcp sport 53 accept


        # Red → DMZ
        ip saddr 192.168.62.43 ip daddr 172.30.128.0/28 accept

        ip saddr 172.30.0.0/17 ip daddr 172.10.10.0/24 accept
        ip saddr 172.10.10.0/24 ip daddr 172.30.0.0/17 accept

        # Internal network ↔ DMZ
        ip saddr 172.30.0.0/17 ip daddr 172.30.128.0/28 accept
        ip saddr 172.30.128.0/28 ip daddr 172.30.0.0/17 accept

        # DMZ web server → internal DB
        ip saddr 172.30.128.10 ip daddr 172.30.0.15 tcp dport 3306 accept

        # DMZ ↔ home network (remote_employee)
        ip saddr 172.30.128.0/28 ip daddr 172.10.10.0/24 accept
        ip saddr 172.10.10.0/24 ip daddr 172.30.128.0/28 accept

        # Allow ALL networks → DMZ web server
        ip daddr 172.30.128.10 tcp dport {80,443} accept
        ip daddr 172.30.128.7 tcp dport 53 accept
        ip daddr 172.30.128.7 udp dport 53 accept
        ip saddr 172.30.128.7 tcp dport 53 accept
        ip saddr 172.30.128.7 udp dport 53 accept
    }
    chain output {
        type filter hook output priority 0; policy accept;  # Allow all outgoing
    }
}

# Commands
- sudo nft flush ruleset
- sudo vi /etc/sysconfig/nftables.conf
- sudo nft -f /etc/sysconfig/nftables.conf
- sudo nft list ruleset
